  <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/theme.min.css" rel="stylesheet">
  <script src="https://cdn.jsdelivr.net/npm/mammoth@1.6.0/mammoth.browser.min.js"></script>
</head>
<body class="bg-light">

  <div class="container my-5">
    <div class="card shadow-sm p-4 mb-4 bg-white rounded">
      <h2 class="text-primary mb-3">Automatizador de Actividades Didácticas (Vue)</h2>
      <p class="text-muted">Carga tu archivo base <code>Actividad.vue</code> y el documento de Word <code>.docx</code>. La herramienta procesará los textos e inyectará los nuevos datos manteniendo tus imágenes intactas.</p>
      
      <div class="row g-3">
        <div class="col-md-6">
          <label class="form-label fw-bold">1. Seleccionar archivo Actividad.vue base:</label>
          <input type="file" id="fileVue" class="form-control" accept=".vue">
        </div>
        
        <div class="col-md-6">
          <label class="form-label fw-bold">2. Seleccionar archivo de Word (.docx):</label>
          <input type="file" id="fileDocx" class="form-control" accept=".docx">
        </div>
      </div>

      <div class="mt-4 text-center">
        <button id="btnProcesar" class="btn btn-success btn-lg px-5" disabled>Procesar Archivos</button>
      </div>
    </div>

    <div id="seccionResultado" class="card shadow-sm p-4 bg-white rounded d-none">
      <div class="d-flex justify-content-between align-items-center mb-3">
        <h4 class="text-success m-0">🎉 ¡Archivo generado con éxito!</h4>
        <div>
          <button id="btnCopiar" class="btn btn-outline-secondary btn-sm me-2">Copiar Código</button>
          <button id="btnDescargar" class="btn btn-primary btn-sm">Descargar Actividad.vue</button>
        </div>
      </div>
      
      <div class="mb-3">
        <label class="form-label fw-bold">Vista previa del código generado:</label>
        <textarea id="txtResultado" class="form-control font-monospace" rows="15" style="font-size: 0.85rem;" readonly></textarea>
      </div>
    </div>
  </div>

  <script>
    let contenidoVueOriginal = '';
    let datosDocxExtraidos = '';

    const inputVue = document.getElementById('fileVue');
    const inputDocx = document.getElementById('fileDocx');
    const btnProcesar = document.getElementById('btnProcesar');
    const seccionResultado = document.getElementById('seccionResultado');
    const txtResultado = document.getElementById('txtResultado');
    const btnDescargar = document.getElementById('btnDescargar');
    const btnCopiar = document.getElementById('btnCopiar');

    // Habilitar botón de procesar solo cuando ambos archivos estén cargados
    function validarInputs() {
      if (inputVue.files.length > 0 && inputDocx.files.length > 0) {
        btnProcesar.disabled = false;
      }
    }

    // Leer el archivo .vue como texto plano
    inputVue.addEventListener('change', function(e) {
      const archivo = e.target.files[0];
      if (!archivo) return;
      const lector = new FileReader();
      lector.onload = function(evt) {
        contenidoVueOriginal = evt.target.result;
        validarInputs();
      };
      lector.readAsText(archivo);
    });

    // Leer el archivo .docx usando Mammoth
    inputDocx.addEventListener('change', function(e) {
      const archivo = e.target.files[0];
      if (!archivo) return;
      const lector = new FileReader();
      lector.onload = function(evt) {
        const arrayBuffer = evt.target.result;
        mammoth.extractRawText({ arrayBuffer: arrayBuffer })
          .then(function(resultado) {
            datosDocxExtraidos = resultado.value;
            validarInputs();
          })
          .catch(function(err) {
            alert('Error al leer el archivo Word: ' + err.message);
          });
      };
      lector.readAsArrayBuffer(archivo);
    });

    // Lógica de procesamiento idéntica a tu script de Node
    btnProcesar.addEventListener('click', function() {
      try {
        // 1. Extraer imágenes del .vue original
        const regexImagenes = /imagen:\s*(require\([^)]+\))/g;
        const imagenesPreservadas = [];
        let matchImg;
        while ((matchImg = regexImagenes.exec(contenidoVueOriginal)) !== null) {
          imagenesPreservadas.push(matchImg[1]);
        }

        // 2. Procesar líneas del Word
        const lineas = datosDocxExtraidos.split(/\r?\n/).map(l => l.trim()).filter(l => l.length > 0);
        
        let tema = '';
        let introduccion = '';
        let preguntasNuevas = [];
        let preguntaActual = null;

        for (let i = 0; i < lineas.length; i++) {
          const linea = lineas[i];

          if (linea.toLowerCase().startsWith('nombre de la actividad')) {
            tema = linea.includes(',') ? linea.substring(linea.indexOf(',') + 1).trim() : lineas[++i].trim();
            tema = tema.replace(/^"|"$/g, '');
            continue;
          }

          if (linea.toLowerCase().startsWith('objetivo de la actividad')) {
            introduccion = linea.includes(',') ? linea.substring(linea.indexOf(',') + 1).trim() : lineas[++i].trim();
            introduccion = introduccion.replace(/^"|"$/g, '');
            continue;
          }

          if (/^pregunta\s*\d+/i.test(linea)) {
            if (preguntaActual) preguntasNuevas.push(preguntaActual);
            let textoPregunta = linea.includes(',') ? linea.substring(linea.indexOf(',') + 1).trim() : lineas[++i].trim();
            textoPregunta = textoPregunta.replace(/^"|"$/g, '');
            preguntaActual = { texto: textoPregunta, opciones: [] };
            continue;
          }

          if (preguntaActual && /^opción\s*[a-e]\)/i.test(linea)) {
            const esCorrecta = /,x$/i.test(linea) || linea.toLowerCase().endsWith(',x"') || linea.toLowerCase().endsWith('x');
            let textoOpcion = linea
              .replace(/^opción\s*[a-e]\)\s*,?/i, '')
              .replace(/,x$/i, '')
              .replace(/,x"$/i, '')
              .replace(/x$/i, '')
              .replace(/^"|"$/g, '')
              .trim();

            if (!textoOpcion && lineas[i + 1] && !/^opción\s*[a-e]\)/i.test(lineas[i + 1]) && !lineas[i + 1].toLowerCase().startsWith('comentario')) {
              textoOpcion = lineas[++i].trim().replace(/^"|"$/g, '');
            }

            preguntaActual.opciones.push({ texto: textoOpcion, esCorrecta });
            continue;
          }
        }
        if (preguntaActual) preguntasNuevas.push(preguntaActual);

        if (introduccion && !introduccion.startsWith('<b> Objetivo:</b>')) {
          introduccion = `<b> Objetivo:</b> ${introduccion}`;
        }

        // 3. Generar el bloque formateado de preguntas
        const letrasId = ['a', 'b', 'c', 'd', 'e'];
        const preguntasFormateadas = preguntasNuevas.map((preg, index) => {
          const imagenOriginal = imagenesPreservadas[index] || "require('@/assets/actividad/imagen1.png')";
          
          const opcionesJS = preg.opciones.map((opc, oIdx) => {
            return `            {
              id: '${letrasId[oIdx] || oIdx}',
              texto: '${opc.texto.replace(/'/g, "\\'")}',
              esCorrecta: ${opc.esCorrecta},
            }`;
          }).join(',\n');

          return `        {
          id: ${index + 1},
          texto: '${preg.texto.replace(/'/g, "\\'")}',
          imagen: ${imagenOriginal},
          barajarRespuestas: true,
          opciones: [
\n${opcionesJS}
          ],
          mensaje_correcto: '¡Muy bien! Ha acertado la respuesta.',
          mensaje_incorrecto: 'Lo sentimos, su respuesta no es la correcta.',
        }`;
        }).join(',\n');

        // 4. Inyectar los datos mediante la expresión regular
        const regexCuestionario = /(cuestionario:\s*\{)([\s\S]*?)(mensaje_final_reprobado:[\s\S]*?\n\s*\},)/;

        const nuevoBloqueCuestionario = `cuestionario: {
      tema: '${tema.replace(/'/g, "\\'")}',
      titulo: 'Cuestionario',
      introduccion:
        '${introduccion.replace(/'/g, "\\'")}',
      barajarPreguntas: true,
      titulo_aprobado: '¡BUEN TRABAJO!',
      titulo_reprobado: 'VUELVA A INTENTARLO.',
      preguntas: [
${preguntasFormateadas}
      ],
      mensaje_final_aprobado: '¡Excelente! Ha superado la actividad.',
      mensaje_final_reprobado:
        'Le recomendamos volver a revisar el componente formativo e intentar nuevamente la actividad didáctica.',
    },`;

        const resultadoFinal = contenidoVueOriginal.replace(regexCuestionario, nuevoBloqueCuestionario);

        // Muestre los resultados en pantalla
        txtResultado.value = resultadoFinal;
        seccionResultado.classList.remove('d-none');
        window.scrollTo({ top: seccionResultado.offsetTop, behavior: 'smooth' });

      } catch (error) {
        alert('❌ Error al procesar los archivos. Revisa el formato del documento.');
        console.error(error);
      }
    });

    // Botón para copiar código al portapapeles
    btnCopiar.addEventListener('click', function() {
      txtResultado.select();
      document.execCommand('copy');
      alert('📋 ¡Código copiado al portapapeles!');
    });

    // Botón para descargar el archivo Actividad.vue directamente
    btnDescargar.addEventListener('click', function() {
      const texto = txtResultado.value;
      const blob = new Blob([texto], { type: 'text/plain;charset=utf-8' });
      const elemento = document.createElement('a');
      elemento.href = URL.createObjectURL(blob);
      elemento.download = 'Actividad.vue';
      elemento.style.display = 'none';
      document.body.appendChild(elemento);
      elemento.click();
      document.body.removeChild(elemento);
    });
  </script>
</body>
